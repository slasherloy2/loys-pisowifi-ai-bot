        return COVERAGE_MSG
    if any(k in text for k in ["install", "taod", "pataod", "libre", "schedule"]):
        return INSTALL_MSG
    if any(k in text for k in ["bayad", "payment", "gcash", "bpi", "bayaran"]):
        return PAYMENT_MSG
    if any(k in text for k in ["hinay", "wala", "putol", "down", "wala net", "lag", "speed", "problema", "support", "tabang", "guba"]):
        return SUPPORT_MSG

@app.route("/", methods=["GET"])
def home():
    return "LOYS Pisowifi AI Bot is ONLINE - Bisaya 24/7"

@app.route("/webhook", methods=["GET"])
def verify():
    mode = request.args.get("hub.mode")
    token = request.args.get("hub.verify_token")
    challenge = request.args.get("hub.challenge")
    if mode == "subscribe" and token == VERIFY_TOKEN:
        return challenge, 200
    return "Verification token mismatch", 403

@app.route("/webhook", methods=["POST"])
def webhook():
    data = request.get_json()
    if data.get("object") == "page":
        for entry in data.get("entry", []):
            for event in entry.get("messaging", []):
                sender_id = event["sender"]["id"]
                if "message" in event and "text" in event["message"]:
                    msg = event["message"]["text"]
                    reply = get_reply(msg)
                    bot.send_text_message(sender_id, reply)
    return "EVENT_RECEIVED", 200
